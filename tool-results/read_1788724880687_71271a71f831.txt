   216→function MyPosts({
   217→  posts,
   218→  onOpenApplications,
   219→  settings,
   220→}: {
   221→  posts: ReceiverPost[]
   222→  onOpenApplications: (post: ReceiverPost) => void
   223→  settings?: PlatformSettings
   224→}) {
   225→  const queryClient = useQueryClient()
   226→
   227→  const cancelMutation = useMutation({
   228→    mutationFn: (id: string) => apiPatch<{ message: string }>(`/api/posts/${id}`, { status: 'CANCELLED' }),

   254→            <CardContent className="space-y-3 p-5">
   255→              <div className="flex items-start justify-between gap-2">
   256→                <div>
   257→                  <p className="font-bold">{post.title}</p>
   258→                  <p className="text-sm text-muted-foreground">

   261→                  </p>
   262→                </div>
   263→                <StatusBadge status={post.status} labels={POST_STATUS_LABELS} />
   264→              </div>
   265→
   266→              <div className="grid grid-cols-2 gap-2 text-xs">
   267→                <div className="rounded-lg bg-secondary/60 p-2.5">
   268→                  <p className="text-muted-foreground">قيمة التكليف</p>

   272→                </div>
   273→                <div className="rounded-lg bg-secondary/60 p-2.5">

   282→                </div>
   283→                <div className="rounded-lg bg-secondary/60 p-2.5">
   284→                  <p className="text-muted-foreground">تاريخ البدء</p>
   285→                  <p className="mt-0.5 font-bold">{formatDate(post.startDate)}</p>
   286→                </div>
   287→                <div className="rounded-lg bg-secondary/60 p-2.5">
   288→                  <p className="text-muted-foreground">الموقع (تلقائي من الجهة)</p>
   289→                  <p className="mt-0.5 font-bold">{post.location ?? 'غير محدد'}</p>
   290→                </div>

   293→              <div className="flex flex-wrap gap-2 text-xs">

   297→                </Badge>
   298→                <Badge variant="outline" className="gap-1">
   299→                  <BadgeCheck className="size-3" />
   300→                  {approvedCount} / {post.nursesNeeded} كادر معتمد
   301→                </Badge>
   302→                {post.hours ? (
   303→                  <Badge variant="outline" className="gap-1">
   304→                    <Clock className="size-3" />
   305→                    {post.hours} ساعة
   306→                  </Badge>
   307→                ) : null}
   308→                <Badge variant="outline" className="gap-1">
   309→                  <UserRound className="size-3" />

   311→                </Badge>
   312→              </div>
   313→

   315→                <Button className="w-full gap-2" onClick={() => onOpenApplications(post)}>
   316→                  <Users className="size-4" />

   326→              {post.status !== 'OPEN' && post.status !== 'CANCELLED' && settings && (
   327→                <PaymentCard
   328→                  settings={settings}
   329→                  breakdown={[
   330→                    { label: `نسبة الإدارة (${settings.adminPercentage}٪)`, amount: Math.round((post.value * settings.adminPercentage) / 100) },
   331→                    { label: 'رسوم التقديم (لكل كادر معتمد)', amount: settings.applicationFee * Math.max(1, approvedCount) },

   333→                  dueAmount={
   334→                    Math.round((post.value * settings.adminPercentage) / 100) +
   335→                    settings.applicationFee * Math.max(1, approvedCount)
   336→                  }
   337→                />
   338→              )}
   339→
   340→              {post.status === 'OPEN' && (
   341→                <Button
   342→                  variant="ghost"
   343→                  size="sm"
   344→                  className="w-full text-muted-foreground hover:text-destructive"
   345→                  disabled={cancelMutation.isPending}
   346→                  onClick={() => cancelMutation.mutate(post.id)}
   347→                >
   348→                  <XCircle className="size-3.5" />
   349→                  إلغاء الإعلان
   350→                </Button>
   351→              )}
   352→            </CardContent>
   353→          </Card>
   354→        )
   355→      })}
   356→    </div>
   357→  )
   358→}

   362→function ApplicationsReview({ postId, nursesNeeded }: { postId: string; nursesNeeded: number }) {
   363→  const queryClient = useQueryClient()
   364→
   365→  const { data, isLoading } = useQuery({
   366→    queryKey: ['post-applications', postId],
   367→    queryFn: () => apiFetcher<{ applications: ApplicantData[] }>(`/api/posts/${postId}/applications`),
   368→  })
   369→
   370→  const reviewMutation = useMutation({
   371→    mutationFn: ({ id, action, note }: { id: string; action: 'APPROVE' | 'REJECT'; note?: string }) =>
   372→      apiPatch<{ message: string }>(`/api/applications/${id}`, { action, note: note ?? '' }),
   373→    onSuccess: (res) => {
   374→      toast.success(res.message)
   375→      queryClient.invalidateQueries({ queryKey: ['post-applications', postId] })
   376→      queryClient.invalidateQueries({ queryKey: ['my-posts'] })
   377→      queryClient.invalidateQueries({ queryKey: ['my-assignments'] })
   378→      queryClient.invalidateQueries({ queryKey: ['stats'] })
   379→    },
   380→    onError: (e: Error) => toast.error(e.message),
   381→  })
   382→
   383→  if (isLoading) return <p className="py-8 text-center text-sm text-muted-foreground">جارٍ تحميل التقديمات...</p>
   384→
   385→  const applications = data?.applications ?? []
   386→  if (applications.length === 0) {
   387→    return (
   388→      <p className="rounded-xl border border-dashed p-8 text-center text-sm text-muted-foreground">
   389→        لا توجد تقديمات بعد — سيصلك إشعار فور تقديم أي كادر على هذا التكليف
   390→      </p>
   391→    )
   392→  }
   393→
   394→  return (
   395→    <div className="space-y-4">
   396→      <p className="rounded-xl bg-secondary/60 px-3 py-2 text-xs text-muted-foreground">
   397→        المطلوب: {nursesNeeded} كادر — عند اكتمال العدد يُغلق التكليف تلقائياً وتُرفض بقية التقديمات
   398→      </p>
   399→      {applications.map((applicant) => (
   400→        <ApplicantCV
   401→          key={applicant.applicationId}
   402→          applicant={applicant}
   403→          reviewing={reviewMutation.isPending}
   404→          onReview={(id, action, note) => reviewMutation.mutate({ id, action, note })}
   405→        />
   406→      ))}
   407→    </div>
   408→  )
   409→}

   545→// ---------- إنشاء تكليف مُعلن ----------
   546→
   547→function CreatePostDialog({
   548→  open,
   549→  onOpenChange,
   550→  nextNumber,
   551→}: {
   552→  open: boolean
   553→  onOpenChange: (open: boolean) => void
   554→  nextNumber: number
   555→}) {
   556→  const queryClient = useQueryClient()
   557→  const [selectedHospital, setSelectedHospital] = useState<Hospital | null>(null)
   558→
   559→  const form = useForm<CreatePostFormValues, unknown, CreatePostInput>({
   560→    resolver: zodResolver(createPostSchema),
   561→    defaultValues: {
   562→      title: '',
   563→      description: '',
   564→      hospitalId: '',
   565→      department: '',
   566→      location: '',
   567→      startDate: '',
   568→      nursesNeeded: '1',
   569→      hours: '',
   570→      gender: 'ANY',
   571→      value: '',
   572→    },
   573→  })
   574→
   575→  // الجهات الصحية والأقسام — تُدار من حساب الإدارة
   576→  const { data: hospitalsData } = useQuery({
   577→    queryKey: ['hospitals'],
   578→    queryFn: () => apiFetcher<{ hospitals: Hospital[] }>('/api/hospitals'),
   579→    enabled: open,
   580→  })
   581→  const { data: departmentsData } = useQuery({
   582→    queryKey: ['departments'],
   583→    queryFn: () => apiFetcher<{ departments: Department[] }>('/api/departments'),
   584→    enabled: open,
   585→  })
   586→  const hospitals = hospitalsData?.hospitals ?? []
   587→  const departments = departmentsData?.departments ?? []
   588→
   589→  const createMutation = useMutation({
   590→    mutationFn: (values: CreatePostInput) => apiPost<{ message: string }>('/api/posts', values),
   591→    onSuccess: (res) => {
   592→      toast.success(res.message)
   593→      queryClient.invalidateQueries({ queryKey: ['my-posts'] })
   594→      queryClient.invalidateQueries({ queryKey: ['open-posts'] })
   595→      queryClient.invalidateQueries({ queryKey: ['stats'] })
   596→      onOpenChange(false)
   597→      form.reset()
   598→      setSelectedHospital(null)
   599→    },
   600→    onError: (e: Error) => toast.error(e.message),
   601→  })
   602→
   603→  const onSubmit = (values: CreatePostInput) => {
   604→    createMutation.mutate(values)
   605→  }
   606→
   607→  return (
   608→    <Dialog open={open} onOpenChange={onOpenChange}>
   609→      <DialogContent className="max-h-[90vh] max-w-2xl overflow-y-auto">
   610→        <DialogHeader>
   611→          <DialogTitle>إنشاء تكليف جديد — التكليف رقم {nextNumber}</DialogTitle>
   612→          <DialogDescription>
   613→            سيظهر العنوان تلقائياً «التكليف رقم {nextNumber}» (ويمكن الإدارة تعديله لاحقاً) — الجهة
   614→            الصحية تُختار من مستشفيات الإدارة والموقع يُعبأ تلقائياً — بدون تاريخ انتهاء
   615→          </DialogDescription>
   616→        </DialogHeader>
   617→        <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4" noValidate>
   618→          <div className="space-y-2">
   619→            <Label htmlFor="p-title">عنوان التكليف (اختياري)</Label>
   620→            <Input
   621→              id="p-title"
   622→              placeholder={`اتركه فارغاً ليكون: التكليف رقم ${nextNumber}`}
   623→              {...form.register('title')}
   624→            />
   625→            {form.formState.errors.title && (
   626→              <p className="text-xs text-destructive">{form.formState.errors.title.message}</p>
   627→            )}
   628→          </div>
   629→
   630→          <div className="grid gap-4 sm:grid-cols-2">
   631→            <div className="space-y-2">
   632→              <Label>الجهة الصحية (من مستشفيات الإدارة)</Label>
   633→              <Select
   634→                onValueChange={(v) => {
   635→                  const h = hospitals.find((x) => x.id === v) ?? null
   636→                  setSelectedHospital(h)
   637→                  form.setValue('hospitalId', v, { shouldValidate: true })
   638→                  // الموقع الفعلي يُعبأ تلقائياً بحسب الجهة الصحية
   639→                  form.setValue('location', h?.location ?? '')
   640→                }}
   641→              >
   642→                <SelectTrigger>
   643→                  <SelectValue placeholder="اختر المستشفى" />
   644→                </SelectTrigger>
   645→                <SelectContent>
   646→                  {hospitals.map((h) => (
   647→                    <SelectItem key={h.id} value={h.id}>
   648→                      {h.name}
   649→                      {h.location ? ` — ${h.location}` : ''}
   650→                    </SelectItem>
   651→                  ))}
   652→                  {hospitals.length === 0 && (
   653→                    <SelectItem value="none" disabled>
   654→                      لا توجد مستشفيات — تُضاف من حساب الإدارة
   655→                    </SelectItem>
   656→                  )}
   657→                </SelectContent>
   658→              </Select>
   659→              {form.formState.errors.hospitalId && (
   660→                <p className="text-xs text-destructive">
   661→                  {form.formState.errors.hospitalId.message}
   662→                </p>
   663→              )}
   664→            </div>
   665→            <div className="space-y-2">
   666→              <Label>القسم (من قوائم الإدارة)</Label>
   667→              <Select onValueChange={(v) => form.setValue('department', v)}>
   668→                <SelectTrigger>
   669→                  <SelectValue placeholder="اختر القسم" />
   670→                </SelectTrigger>
   671→                <SelectContent>
   672→                  {departments.map((d) => (
   673→                    <SelectItem key={d.id} value={d.name}>
   674→                      {d.name}
   675→                    </SelectItem>
   676→                  ))}
   677→                  {departments.length === 0 && (
   678→                    <SelectItem value="none" disabled>
   679→                      لا توجد أقسام — تُضاف من حساب الإدارة
   680→                    </SelectItem>
   681→                  )}
   682→                </SelectContent>
   683→              </Select>
   684→            </div>
   685→          </div>
   686→
   687→          {selectedHospital?.location && (
   688→            <p className="flex items-center gap-2 rounded-xl bg-emerald-50 px-3 py-2 text-xs font-bold text-emerald-700">
   689→              <MapPin className="size-3.5" />
   690→              الموقع الفعلي (تلقائي): {selectedHospital.location}
   691→            </p>
   692→          )}
   693→
   694→          <div className="grid gap-4 sm:grid-cols-3">
   695→            <div className="space-y-2">
   696→              <Label htmlFor="p-value">قيمة التكليف (ريال)</Label>
   697→              <Input
   698→                id="p-value"
   699→                type="number"
   700→                min={1}
   701→                placeholder="مثال: 120000"
   702→                {...form.register('value')}
   703→              />
   704→              {form.formState.errors.value && (
   705→                <p className="text-xs text-destructive">{form.formState.errors.value.message}</p>
   706→              )}
   707→            </div>
   708→            <div className="space-y-2">
   709→              <Label htmlFor="p-hours">عدد الساعات</Label>
   710→              <Input
   711→                id="p-hours"
   712→                type="number"
   713→                min={1}
   714→                max={999}
   715→                placeholder="مثال: 8"
   716→                {...form.register('hours')}
   717→              />
   718→              {form.formState.errors.hours && (
   719→                <p className="text-xs text-destructive">{form.formState.errors.hours.message}</p>
   720→              )}
   721→            </div>
   722→            <div className="space-y-2">
   723→              <Label htmlFor="p-needed">عدد الكادر المطلوب</Label>
   724→              <Input id="p-needed" type="number" min={1} max={50} {...form.register('nursesNeeded')} />
   725→              {form.formState.errors.nursesNeeded && (
   726→                <p className="text-xs text-destructive">
   727→                  {form.formState.errors.nursesNeeded.message}
   728→                </p>
   729→              )}
   730→            </div>
   731→          </div>
   732→
   733→          <div className="grid gap-4 sm:grid-cols-2">
   734→            <div className="space-y-2">
   735→              <Label>الجنس المطلوب</Label>
   736→              <Select onValueChange={(v) => form.setValue('gender', v as CreatePostInput['gender'])}>
   737→                <SelectTrigger>
   738→                  <SelectValue placeholder="اختر الجنس" />
   739→                </SelectTrigger>
   740→                <SelectContent>
   741→                  <SelectItem value="ANY">أي جنس</SelectItem>
   742→                  <SelectItem value="MALE">ذكر</SelectItem>
   743→                  <SelectItem value="FEMALE">أنثى</SelectItem>
   744→                </SelectContent>
   745→              </Select>
   746→            </div>
   747→            <div className="space-y-2">
   748→              <Label htmlFor="p-start">تاريخ البدء</Label>
   749→              <Input id="p-start" type="date" {...form.register('startDate')} />
   750→              {form.formState.errors.startDate && (
   751→                <p className="text-xs text-destructive">
   752→                  {form.formState.errors.startDate.message}
   753→                </p>
   754→              )}
   755→            </div>
   756→          </div>
   757→
   758→          <div className="space-y-2">
   759→            <Label htmlFor="p-desc">وصف التكليف (اختياري)</Label>
   760→            <Textarea
   761→              id="p-desc"
   762→              rows={3}
   763→              placeholder="تفاصيل المهام والمسؤوليات والوردية..."
   764→              {...form.register('description')}
   765→            />
   766→          </div>
   767→
   768→          <DialogFooter>
   769→            <Button type="button" variant="outline" onClick={() => onOpenChange(false)}>
   770→              إلغاء
   771→            </Button>
   772→            <Button type="submit" disabled={createMutation.isPending} className="gap-2">
   773→              <CalendarDays className="size-4" />
   774→              {createMutation.isPending ? 'جارٍ النشر...' : 'نشر التكليف'}
   775→            </Button>
   776→          </DialogFooter>
   777→        </form>
   778→      </DialogContent>
   779→    </Dialog>
   780→  )
   781→}
